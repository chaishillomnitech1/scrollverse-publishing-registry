pragma solidity ^0.8.20;

    event ZakatDistributed(address indexed recipient, uint256 amount);
    event FrequencyEncoded(uint256 frequency);
    
    constructor(address _nftContract, address _zakatWallet) ERC20("Mirror Token", "MIRROR") {
        nftContract = _nftContract;
        zakatWallet = _zakatWallet;
        _mint(msg.sender, TOTAL_SUPPLY);

        emit FrequencyEncoded(963);

        lastDividendClaim[msg.sender] = block.timestamp;
        
        _mint(msg.sender, dividend);
        emit ConsciousnessDividend(msg.sender, dividend);
    }
}
